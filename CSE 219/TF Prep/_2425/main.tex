\chapter*{How to read this book}
\begin{tabularx}{\textwidth}{@{}>{\sffamily\bfseries\small}p{3.2cm}X@{}}
\textcolor{navy}{The setting} & The story or background shared by all parts of a question, copied from the paper.\\[3pt]
\textcolor{coral}{Question card} & Each sub-question with its marks badge, exactly as set.\\[3pt]
\textcolor{teal}{Solution} & Step-by-step working: the idea first, then the calculation, then a check.\\[3pt]
\textcolor{plum}{Key idea} & The one principle the examiner is testing --- worth remembering for other papers.\\[3pt]
\textcolor{notebd}{Note} & Where the question allows a choice (a sketch parameter, a convention), the choice is stated here.\\[3pt]
\colorbox{ansbg}{\textcolor{teal}{Answer}} & The final result, boxed, ready to copy into a script.\\
\end{tabularx}

\medskip
\begin{keyidea}Section~A questions are long ``application'' stories. Read the setting once, translate every sentence into a signal-and-system statement (input, system, output, sampling rate, transform), and the sub-questions become routine.\end{keyidea}

\medskip
{\sffamily\small\bfseries\color{navy}Conventions used throughout}
\begin{itemize}[leftmargin=1.4em,itemsep=1pt]
\item $\omega=2\pi f$; $W_N=e^{-j2\pi/N}$; $X[k]=\sum_{n=0}^{N-1}x[n]W_N^{kn}$.
\item Fourier series $x(t)=\sum_k a_ke^{jk\omega_0t}$, $a_k=\frac1T\int_Tx(t)e^{-jk\omega_0t}\dd t$.
\item Fourier transform $X(j\omega)=\int x(t)e^{-j\omega t}\dd t$; Laplace $X(s)=\int x(t)e^{-st}\dd t$; $z$-transform $X(z)=\sum x[n]z^{-n}$.
\end{itemize}
